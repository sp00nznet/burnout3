cmake_minimum_required(VERSION 3.20)
project(burnout3_game C CXX)

# Burnout 3: Takedown - Recompiled game executable

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ── Dear ImGui (vendored) ──────────────────────────────────────
set(IMGUI_DIR "${CMAKE_SOURCE_DIR}/third_party/imgui")
if(WIN32)
    set(IMGUI_SOURCES
        ${IMGUI_DIR}/imgui.cpp
        ${IMGUI_DIR}/imgui_draw.cpp
        ${IMGUI_DIR}/imgui_tables.cpp
        ${IMGUI_DIR}/imgui_widgets.cpp
        ${IMGUI_DIR}/imgui_demo.cpp
        ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp
        ${IMGUI_DIR}/backends/imgui_impl_win32.cpp
    )
    set(MENU_GUI_SRC menu_gui.cpp)
else()
    # Linux first cut: skip ImGui entirely; menu_gui is a stub that does
    # nothing. Wire imgui_impl_sdl2 + imgui_impl_opengl3 in a follow-up.
    set(IMGUI_SOURCES "")
    set(MENU_GUI_SRC menu_gui_stub.c)
endif()

# ── Recompiled function sources (auto-generated) ──────────────
# Generated by: py -3 -m tools.recomp <xbe> -c data_init --split 1000
file(GLOB RECOMP_GEN_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen/recomp_*.c"
)

# ── NV2A GPU emulation (from xemu) ──────────────────────────────
set(NV2A_DIR "${CMAKE_SOURCE_DIR}/src/nv2a")
set(NV2A_SOURCES
    ${NV2A_DIR}/nv2a_core.c
    ${NV2A_DIR}/nv2a_mmio_hook.c
    ${NV2A_DIR}/nv2a_pb_test.c
    ${NV2A_DIR}/nv2a_pgraph_d3d11.c
    ${NV2A_DIR}/nv2a_pb_replay.c
)

# Game executable sources
set(GAME_SOURCES
    main.c
    txd_loader.c
    bgv_loader.c
    track_loader.c
    static_textures.c
    rw_renderer.c
    video_player.c
    awd_loader.c
    rw_bridge.c
    fe_menu.c
    ${MENU_GUI_SRC}
    recomp/recomp_test.c
    recomp/recomp_manual.c
    ${NV2A_SOURCES}
    ${RECOMP_GEN_SOURCES}
    ${IMGUI_SOURCES}
)

# Executable: WIN32 keyword on Windows (GUI subsystem); plain on POSIX.
if(WIN32)
    add_executable(burnout3 WIN32 ${GAME_SOURCES})
else()
    add_executable(burnout3 ${GAME_SOURCES})
endif()

# Link compatibility layers
target_link_libraries(burnout3 PRIVATE
    xbox_kernel
    d3d8_compat
    dsound_compat
    input_compat
    apu_emu
)

# System libraries
if(WIN32)
    target_link_libraries(burnout3 PRIVATE
        kernel32 user32 gdi32
        mfplat mfreadwrite mfuuid mf
        ole32 propsys
        d3d11 dxgi
    )
else()
    find_package(SDL2 REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(burnout3 PRIVATE
        SDL2::SDL2 Threads::Threads m
    )
endif()

# Include paths
target_include_directories(burnout3 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/recomp
    ${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen
    ${CMAKE_SOURCE_DIR}/src
    ${NV2A_DIR}
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)

# Compiler options
if(MSVC)
    target_compile_options(burnout3 PRIVATE
        /W4
        /WX-
        /O2
    )
    # Reduce warning noise on generated/recompiled code and ImGui
    set_source_files_properties(${RECOMP_GEN_SOURCES} recomp/recomp_manual.c PROPERTIES
        COMPILE_FLAGS "/w"
    )
    set_source_files_properties(${IMGUI_SOURCES} PROPERTIES
        COMPILE_FLAGS "/w"
    )
    set_source_files_properties(${NV2A_SOURCES} PROPERTIES
        COMPILE_FLAGS "/w"
    )
    target_compile_definitions(burnout3 PRIVATE
        _CRT_SECURE_NO_WARNINGS
    )
    # Disable ASLR so we can map Xbox memory at fixed addresses
    # Large stack: recompiled code uses deep native call chains
    # (each Xbox call/ret becomes a C function call/return)
    target_link_options(burnout3 PRIVATE
        /DYNAMICBASE:NO
        /FIXED:NO
        /MAP
        /DEBUG
        /STACK:16777216
    )
elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(burnout3 PRIVATE
        -Wall -Wextra -Wno-unused-parameter -O2
    )
    # The auto-generated recompiled code calls a handful of sibling sub_XXX
    # functions that the recompiler did not emit declarations for, and does
    # a lot of int<->pointer punning. Modern gcc treats these as errors;
    # demote them to warnings on gen sources only.
    set_source_files_properties(${RECOMP_GEN_SOURCES} PROPERTIES
        COMPILE_FLAGS "-w -Wno-error=implicit-function-declaration -Wno-error=int-conversion -Wno-error=incompatible-pointer-types"
    )
endif()

# Debug info
target_compile_definitions(burnout3 PRIVATE
    $<$<CONFIG:Debug>:_DEBUG>
)

# Output to project root for easy testing
set_target_properties(burnout3 PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin"
)
